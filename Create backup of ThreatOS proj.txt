# Create backup directory
mkdir -p /backup/threatos

# 1. Database backup
PGPASSWORD=threatos_secret pg_dump -h localhost -U threatos threatos | \
    gzip > /backup/threatos/db_$(date +%Y%m%d).sql.gz

echo "DB backup: $(ls -lh /backup/threatos/db_*.sql.gz | tail -1)"

# 2. Code zip — excludes venv, node_modules, dist, .env
cd /opt
zip -r /backup/threatos/threatos_code_$(date +%Y%m%d).zip threatos/ \
    --exclude "threatos/.venv/*" \
    --exclude "threatos/threatos/frontend/node_modules/*" \
    --exclude "threatos/threatos/frontend/dist/*" \
    --exclude "threatos/.env"

echo "Code backup: $(ls -lh /backup/threatos/threatos_code_*.zip | tail -1)"

# 3. Backup the .env separately to a secure location
# (never include secrets in the code zip)
cp /opt/threatos/.env /backup/threatos/.env_$(date +%Y%m%d).bak
chmod 600 /backup/threatos/.env_*.bak

# 4. Backup systemd service files
cp /etc/systemd/system/threatos-*.service /backup/threatos/
cp /etc/systemd/system/threatos-*.timer   /backup/threatos/
cp /etc/nginx/conf.d/threatos.conf        /backup/threatos/

# 5. List everything
ls -lh /backup/threatos/
